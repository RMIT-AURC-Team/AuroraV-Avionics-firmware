25void A3G4250D_init(A3G4250D *gyro, GPIO_TypeDef *port, unsigned long cs, float sensitivity, const uint8_t *axes, const int8_t *sign) {
26 SPI_init(&gyro->base, SENSOR_GYRO, SPI1, MODE8, port, cs);
27 gyro->sensitivity = sensitivity;
28 gyro->update = A3G4250D_update;
29 gyro->readGyro = A3G4250D_readGyro;
30 gyro->readRawBytes = A3G4250D_readRawBytes;
31 gyro->processRawBytes = A3G4250D_processRawBytes;
32 memcpy(gyro->axes, axes, A3G4250D_DATA_COUNT);
33 memcpy(gyro->sign, sign, A3G4250D_DATA_COUNT);
34
35 const uint32_t superDelay = 0xFFFF;
36 volatile uint8_t counter = 0;
37
38 // Wait for the spefified period - need to wait for 2ms here.
39 for (uint32_t i = 0; i < superDelay; i++) {
40 counter++;
41 }
42
43 A3G4250D_writeRegister(gyro, A3G4250D_CTRL_REG1, A3G4250D_CTRL_REG1_ODR_800Hz | A3G4250D_CTRL_REG1_AXIS_ENABLE | A3G4250D_CTRL_REG1_PD_ENABLE);
44}

86void A3G4250D_processRawBytes(A3G4250D *gyro, uint8_t *bytes, float *out) {
87 out[0] = gyro->sign[0] * gyro->sensitivity * (int16_t)(((uint16_t)bytes[0] << 8) | bytes[1]); // gyro X
88 out[1] = gyro->sign[1] * gyro->sensitivity * (int16_t)(((uint16_t)bytes[2] << 8) | bytes[3]); // gyro Y
89 out[2] = gyro->sign[2] * gyro->sensitivity * (int16_t)(((uint16_t)bytes[4] << 8) | bytes[5]); // gyro Z
90}

101void A3G4250D_readRawBytes(A3G4250D *gyro, uint8_t *out) {
102#define INDEX_AXES(index, byte) 2 * gyro->axes[index] + byte
103 out[INDEX_AXES(0, 0)] = A3G4250D_readRegister(gyro, A3G4250D_OUT_X_H); // gyro X high
104 out[INDEX_AXES(0, 1)] = A3G4250D_readRegister(gyro, A3G4250D_OUT_X_L); // gyro X low
105 out[INDEX_AXES(1, 0)] = A3G4250D_readRegister(gyro, A3G4250D_OUT_Y_H); // gyro Y high
106 out[INDEX_AXES(1, 1)] = A3G4250D_readRegister(gyro, A3G4250D_OUT_Y_L); // gyro Y low
107 out[INDEX_AXES(2, 0)] = A3G4250D_readRegister(gyro, A3G4250D_OUT_Z_H); // gyro Z high
108 out[INDEX_AXES(2, 1)] = A3G4250D_readRegister(gyro, A3G4250D_OUT_Z_L); // gyro Z low
109#undef INDEX_AXES
110}

114void A3G4250D_writeRegister(A3G4250D *gyro, uint8_t address, uint8_t data) {
115 SPI spi = gyro->base;
116
117 spi.port->ODR &= ~spi.cs;
118
119 // Send read command and address
120 uint8_t payload = address & 0x7F; // Load payload with address and read command
121 spi.transmit(&spi, payload); // Transmit payload
122 spi.transmit(&spi, data); // Transmit dummy data and read response data
123
124 spi.port->ODR |= spi.cs;
125}
126
127uint8_t A3G4250D_readRegister(A3G4250D *gyro, uint8_t address) {
128 uint8_t response = 0;
129 SPI spi = gyro->base;
130
131 spi.port->ODR &= ~spi.cs;
132
133 // Send read command and address
134 uint8_t payload = address | 0x80; // Load payload with address and read command
135 response = spi.transmit(&spi, payload); // Transmit payload
136 response = spi.transmit(&spi, 0xFF); // Transmit dummy data and read response data
137
138 spi.port->ODR |= spi.cs;
139
140 return response;
141}
