120void Flash_erase(Flash *flash) {

145void Flash_writePage(Flash *flash, uint32_t address, uint8_t *data) {
146 _Flash_writeEnable(flash);
147 uint8_t status = 0;
148 SPI spi = flash->base;
149
150 spi.port->ODR &= ~spi.cs;
151
152 // Send Page Program instruction and 24-bit address
153 spi.transmit(&spi, FLASH_PAGE_PROGRAM);
154 spi.transmit(&spi, (address & 0xFF0000) >> 16);
155 spi.transmit(&spi, (address & 0xFF00) >> 8);
156 spi.transmit(&spi, (address & 0xFF));
157
158 // Send page data
159 for (int i = 0; i < 256; i++) {
160 spi.transmit(&spi, data[i]);
161 }
162
163 spi.port->ODR |= spi.cs;
164
165 // Wait until chip BUSY is clear
166 do {
167 _Flash_readStatus1(flash, &status);
168 } while (status & 0x1);
169}
