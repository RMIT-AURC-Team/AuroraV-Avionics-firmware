128void Flash_erase(Flash *flash) {
129 _Flash_writeEnable(flash);
130 SPI spi = flash->base;
131 uint8_t status = 0;
132
133 // Send Erase Chip instruction
134 spi.port->ODR &= ~spi.cs;
135 spi.transmit(&spi, FLASH_ERASE_CHIP);
136 spi.port->ODR |= spi.cs;
137
138 // Wait until chip BUSY is clear
139 do {
140 _Flash_readStatus1(flash, &status);
141 } while (status & 0x01);
142}

154void Flash_writePage(Flash *flash, uint32_t address, uint8_t *data) {
155 _Flash_writeEnable(flash);
156 uint8_t status = 0;
157 SPI spi = flash->base;
158
159 spi.port->ODR &= ~spi.cs;
160
161 // Send Page Program instruction and 24-bit address
162 spi.transmit(&spi, FLASH_PAGE_PROGRAM);
163 spi.transmit(&spi, (address & 0xFF0000) >> 16);
164 spi.transmit(&spi, (address & 0xFF00) >> 8);
165 spi.transmit(&spi, (address & 0xFF));
166
167 // Send page data
168 for (int i = 0; i < 256; i++) {
169 spi.transmit(&spi, data[i]);
170 }
171
172 spi.port->ODR |= spi.cs;
173
174 // Wait until chip BUSY is clear
175 do {
176 _Flash_readStatus1(flash, &status);
177 } while (status & 0x1);
178}

190void Flash_readPage(Flash *flash, uint32_t address, volatile uint8_t *data) {
191 SPI spi = flash->base;
192
193 spi.port->ODR &= ~spi.cs;
194
195 // Send Read Data instruction and 24-bit address
196 spi.transmit(&spi, FLASH_READ_DATA);
197 spi.transmit(&spi, (address & 0xFF0000) >> 16);
198 spi.transmit(&spi, (address & 0xFF00) >> 8);
199 spi.transmit(&spi, (address & 0xFF));
200
201 for (int i = 0; i < 256; i++) {
202 data[i] = spi.transmit(&spi, 0x0F);
203 }
204
205 spi.port->ODR |= spi.cs;
206}
