50void _UART_setup(UART *uart) {
51 GPIO_TypeDef *port = uart->port;
52 USART_TypeDef *interface = uart->interface;
53
54 port->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk | GPIO_MODER_MODE8_Msk);
55 port->MODER |= (0x02 << GPIO_MODER_MODE6_Pos) | (0x02 << GPIO_MODER_MODE7_Pos) | (0x2 << GPIO_MODER_MODE8_Pos);
56 port->AFR[0] &= ~0xFF000000; // clears AFRL 6 and 7 and
57 port->AFR[0] |= 0x88000000; // sets PC 6 and 7 to AF8
58 port->PUPDR |= (0x01 << GPIO_PUPDR_PUPD7_Pos);
59
60 port->OSPEEDR &= ~GPIO_OSPEEDR_OSPEED6;
61 port->OSPEEDR |= (0x03 << GPIO_OSPEEDR_OSPEED6_Pos);
62
63 uint16_t usartDiv = 168000000 / ((2 - (uart->over8)) * uart->baud);
64 interface->BRR &= 0xFFFF0000; // clear mantissa and div in baud rate reg
65 interface->BRR |= (usartDiv); // set mantissa and div in baud rate reg to 9600
66
67 interface->CR1 &= ~0x400; // disable parity
68 interface->CR2 &= ~0xE00; // disable synchrnous mode
69 interface->CR3 &= ~0x300; // disable flow control
70 interface->CR1 |= 0x202C; // enable usart, enable receive and transmitt
71}

86void UART_send(UART *uart, uint8_t data) {
87 USART_TypeDef *interface = uart->interface;
88 while ((interface->SR & USART_SR_TXE) == 0);
89 interface->DR = data;
90 while ((interface->SR & USART_SR_TC) == 0);
91}
