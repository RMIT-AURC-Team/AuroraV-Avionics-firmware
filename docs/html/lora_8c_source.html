37void LoRa_init(LoRa *lora, GPIO_TypeDef *port, unsigned long cs, Bandwidth bw, SpreadingFactor sf, CodingRate cr) {
38 SPI_init(&lora->base, COMM_LORA, SPI3, MODE16, port, cs);
39 lora->transmit = LoRa_transmit;
40
41 _LoRa_setMode(lora, SLEEP); // Set mode to sleep
42
43 // Set interrupt pin
44 LoRa_writeRegister(lora, RegDioMapping1, 0x40);
45
46 /* clang-format off */
47 LoRa_writeRegister(lora, LORA_REG_OP_MODE,
48 0x01 << LORA_REG_OP_MODE_LONG_RANGE_Pos // Enable LoRa
49 );
50
51 LoRa_writeRegister(lora, LORA_REG_MODEM_CONFIG1,
52 bw << LORA_REG_MODEM_CONFIG1_BW_Pos // Set bandwidth
53 | cr << LORA_REG_MODEM_CONFIG1_CR_Pos // Set coding rate
54 | 0x00 << LORA_REG_MODEM_CONFIG1_CRC_Pos // Enable CRC
55 );
56 /* clang-format on */
57
59 LoRa_writeRegister(lora, LORA_REG_MODEM_CONFIG2, 0x94);
60
61 // Set payload length
62 LoRa_writeRegister(lora, LORA_REG_PAYLOAD_LENGTH, LORA_MSG_LENGTH);
63 LoRa_writeRegister(lora, LORA_REG_MAX_PAYLOAD_LENGTH, LORA_MSG_LENGTH);
64
65 _LoRa_setMode(lora, STDBY); // Set mode to standby
66}

109LoRa_Packet LoRa_AVData(
110 uint8_t id,
111 uint8_t currentState,
112 uint8_t *lAccelData,
113 uint8_t *hAccelData,
114 uint8_t lenAccel,
115 uint8_t *gyroData,
116 uint8_t lenGyro,
117 float altitude,
118 float velocity
119) {
120 LoRa_Packet msg;
121
122 // Convert altitude float to byte array
123 union {
124 float f;
125 uint8_t b[4];
126 } a;
127 a.f = altitude;
128
129 // Convert velocity float to byte array
130 union {
131 float f;
132 uint8_t b[4];
133 } v;
134 v.f = velocity;
135
136 int idx = 0;
137 // Append to struct data array
138 msg.id = id;
139 msg.data[idx++] = currentState;
140 memcpy(&msg.data[idx], lAccelData, lenAccel);
141 memcpy(&msg.data[idx += lenAccel], hAccelData, lenAccel);
142 memcpy(&msg.data[idx += lenAccel], gyroData, lenGyro);
143 memcpy(&msg.data[idx += lenGyro], a.b, sizeof(float));
144 memcpy(&msg.data[idx += sizeof(float)], v.b, sizeof(float));
145
146 return msg;
147}

177void LoRa_transmit(LoRa *lora, uint8_t *pointerdata) {
178 LoRa_writeRegister(lora, LORA_REG_IRQ_FLAGS, 0x08); // clears the status flags
179 LoRa_writeRegister(lora, LORA_REG_FIFO_ADDR_PTR, 0x80); // set pointer adddress to TX
180
181 // Load data into transmit FIFO
182 for (int i = 0; i < 32; i++) {
183 LoRa_writeRegister(lora, LORA_REG_FIFO, pointerdata[i]);
184 }
185
186 // Set device to transmit
187 _LoRa_setMode(lora, TX);
188
189 // Clear the status flags
190 LoRa_writeRegister(lora, LORA_REG_IRQ_FLAGS, 0x08);
191}

195void LoRa_writeRegister(LoRa *lora, uint8_t address, uint8_t data) {
196 SPI spi = lora->base;
197
198 uint16_t payload = (address << 0x08) | (1 << 0x0F); // Load payload with address and write command
199 payload |= data; // Append data to payload
200 spi.port->ODR &= ~spi.cs; // Lower chip select
201 spi.transmit(&spi, payload); // Send payload over SPI
202 spi.port->ODR |= spi.cs; // Raise chip select
203}
204
205uint8_t LoRa_readRegister(LoRa *lora, uint8_t address) {
206 SPI spi = lora->base;
207 uint16_t response;
208
209 uint16_t payload = (address << 0x08); // Load payload with address and read command
210 spi.port->ODR &= ~spi.cs; // Lower chip select
211 response = spi.transmit(&spi, payload); // Receive payload over SPI
212 spi.port->ODR |= spi.cs; // Raise chip select
213 return (uint8_t)response;
214}
