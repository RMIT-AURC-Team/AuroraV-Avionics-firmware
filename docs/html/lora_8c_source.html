180void LoRa_writeRegister(LoRa *lora, uint8_t address, uint8_t data) {
181 SPI spi = lora->base;
182
183 uint16_t payload = (address << 0x08) | (1 << 0x0F); // Load payload with address and write command
184 payload |= data; // Append data to payload
185 spi.port->ODR &= ~spi.cs; // Lower chip select
186 spi.transmit(&spi, payload); // Send payload over SPI
187 spi.port->ODR |= spi.cs; // Raise chip select
188}
