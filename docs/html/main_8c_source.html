199void vStateUpdate(void *argument) {
200 TickType_t xLastWakeTime;
201 const TickType_t xFrequency = pdMS_TO_TICKS(20); // 50Hz
202
203 unsigned int CANHigh = 0;
204 unsigned int CANLow = 0;
205 unsigned int id = 0;
206
207 for (;;) {
208 // Block until 20ms interval
209 vTaskDelayUntil(&xLastWakeTime, xFrequency);
210
211 // Emergency aerobrakes for excessive tilt
212 if (tilt >= 30.0f) {
213 // CAN payload for aerobrakes retract
214 CANHigh = 0x00000000;
215 CANLow = 0x00000000;
216 id = CAN_HEADER_AEROBRAKES_RETRACT;
217 CAN_TX(2, 8, CANHigh, CANLow, id);
218 }
219
220 switch (currentState) {
221 case PRELAUNCH:
222 if (pAccel_s->accelData[ZINDEX] >= ACCEL_LAUNCH) {
223#ifdef FLIGHT_TEST
224 GPIOB->ODR ^= 0X8000;
225 GPIOD->ODR ^= 0X8000;
226#endif
227#ifndef DEBUG
228 vTaskDelete(xUsbTransmitHandle);
229 vTaskDelete(xUsbReceiveHandle);
230#endif
231 vTaskDelete(xGpsReadHandle);
232 xEventGroupSetBits(xTaskEnableGroup, GROUP_TASK_ENABLE_FLASH); // Enable flash
233 xEventGroupSetBits(xTaskEnableGroup, GROUP_TASK_ENABLE_HIGHRES); // Enable high resolution data acquisition
234 xEventGroupSetBits(xTaskEnableGroup, GROUP_TASK_ENABLE_LOWRES); // Enable low resolution data acquisition
235 currentState = LAUNCH;
236 }
237 break;
238
239 case LAUNCH:
240 avgVel.calculateMovingAverage(&avgVel, &avgVelCurrent);
241 // Send altitude to aerobrakes via CAN
242 CANHigh = 0x00000000;
243 memcpy(&CANLow, &altitude, sizeof(float));
244 id = CAN_HEADER_AEROBRAKES_DATA;
245 CAN_TX(2, 8, CANHigh, CANLow, id);
246 // Transition to motor burnout state on velocity decrease
247 if ((avgVelCurrent - avgVelPrevious) < 0) {
248#ifdef FLIGHT_TEST
249 GPIOB->ODR ^= 0x8000;
250 GPIOD->ODR ^= 0x8000;
251#endif
252 currentState = COAST;
253 }
254 avgVelPrevious = avgVelCurrent;
255 break;
256
257 case COAST:
258 avgPress.calculateMovingAverage(&avgPress, &avgPressCurrent);
259 // Send altitude to aerobrakes via CAN
260 CANHigh = 0x00000000;
261 memcpy(&CANLow, &altitude, sizeof(float));
262 id = CAN_HEADER_AEROBRAKES_DATA;
263 CAN_TX(2, 8, CANHigh, CANLow, id);
264 // Transition to apogee state on three way vote of altitude, velocity, and tilt
265 // apogee is determined as two of three conditions evaluating true
266 if ((((avgPressCurrent - avgPressPrevious) > 0) + (tilt >= 90) + (velocity < 0.0f)) >= 2) {
267#ifdef FLIGHT_TEST
268 GPIOB->ODR ^= 0x8000;
269 GPIOD->ODR ^= 0x8000;
270#endif
271 vTaskDelete(xDataAqcquisitionHHandle);
272 vTaskDelete(xDataAqcquisitionLHandle);
273 vTaskDelete(xLoRaSampleHandle);
274 xTaskCreate(vGpsRead, "GpsRead", 512, NULL, configMAX_PRIORITIES - 6, &xGpsReadHandle);
275 currentState = APOGEE;
276 // Send transmission to trigger apogee E-matches
277 }
278 avgPressPrevious = avgPressCurrent;
279 break;
280
281 case APOGEE:
282 // Retract aerobrakes
283 CANHigh = 0x00000000;
284 CANLow = 0x00000000;
285 id = CAN_HEADER_AEROBRAKES_RETRACT;
286 CAN_TX(2, 8, CANHigh, CANLow, id);
287 // Deploy drogue chute
288 GPIOD->ODR |= 0x8000;
289 // Transition to descent state when below main deployment altitude
290 if (altitude <= MAIN_ALTITUDE_METERS) {
291#ifdef FLIGHT_TEST
292 GPIOB->ODR ^= 0X8000;
293 GPIOD->ODR ^= 0X8000;
294#endif
295 currentState = DESCENT;
296 // Add descent event dataframe to buffer
297 }
298 break;
299
300 case DESCENT:
301 // Handle descent state actions
302 break;
303 }
304 }
305}

334void vFlashBuffer(void *argument) {
335 const TickType_t timeout = pdMS_TO_TICKS(1);
336 uint32_t pageAddr = 0;
337 for (;;) {
338 // Wait for write flag to be ready, clear flag on exit
339 EventBits_t uxBits = xEventGroupWaitBits(xTaskEnableGroup, GROUP_TASK_ENABLE_FLASH, pdTRUE, pdFALSE, timeout);
340 if (uxBits & GROUP_TASK_ENABLE_FLASH) {
341 bool success = mem.readPage(&mem, outBuff); // Flush data to output buffer
342 if (success) {
343 // Write data to flash memory
344 flash.writePage(&flash, pageAddr, outBuff);
345 pageAddr += 0x100;
346 }
347 }
348 }

362void vLoRaTransmit(void *argument) {
363 const TickType_t blockTime = pdMS_TO_TICKS(250);
364 uint8_t rxData[LORA_MSG_LENGTH];
365
366 for (;;) {
367 // Wait for SX1272 to be ready for transmission
368 EventBits_t uxBits = xEventGroupWaitBits(xMsgReadyGroup, GROUP_MESSAGE_READY_LORA, pdFALSE, pdFALSE, blockTime);
369 if ((uxBits & GROUP_MESSAGE_READY_LORA)) {
370
371 // Wait to receive message in buffer
372 size_t xReceivedBytes = xMessageBufferReceive(
373 xLoRaTxBuff,
374 (void *)rxData,
375 sizeof(rxData),
376 blockTime
377 );
378
379 // Transmit if message is available
380 if (xReceivedBytes) {
381 lora.transmit(&lora, rxData);
382 xEventGroupClearBits(xMsgReadyGroup, GROUP_MESSAGE_READY_LORA);
383 }
384 }
385 }

478void vUsbReceive(void *argument) {
479 const TickType_t timeout = pdMS_TO_TICKS(20);
480 uint8_t rxData;
481
482 for (;;) {
483 // Read byte from UART Rx buffer, skip loop if empty
484 if (!xStreamBufferReceive(xUsbRxBuff, (void *)&rxData, 1, timeout))
485 continue;
486
487 // Send byte back for display
488 usb.send(&usb, rxData);
489
490 // Process command and reset buffer on <Enter> input
491 if (rxData == CARRIAGE_RETURN) {
492 usb.print(&usb, "\n"); // Send newline back for display
493 usbRxBuff[usbRxBuffIdx - 1] = '\0'; // Replace carriage return with null terminator
494 usbCommandParse(usbRxBuff); // Parse and execute command
495 usbRxBuffIdx = 0; // Reset buffer
496 }
497
498 // Clear terminal on <Ctrl-c> input
499 else if (rxData == SIGINT) {
500 usbCommandParse((uint8_t *)"clear");
501 usbRxBuffIdx = 0;
502 }
503
504 // Erase character and move cursor backwards on <BS> input
505 else if (rxData == BACKSPACE) {
506 usb.print(&usb, " \b");
507 if (usbRxBuffIdx)
508 usbRxBuffIdx -= 2;
509 }
510 }

548void vDataAcquisitionH(void *argument) {
549 float dt = 0.002;
550
551 TickType_t xLastWakeTime;
552 const TickType_t xFrequency = pdMS_TO_TICKS(2); // 500Hz
553 const TickType_t blockTime = pdMS_TO_TICKS(0);
554
555 for (;;) {
556 // Block until 2ms interval
557 vTaskDelayUntil(&xLastWakeTime, xFrequency);
558
559 // Select which accelerometer to use
560 pAccel_s = (pAccel_s->accelData[ZINDEX] < 15) ? &lAccel_s : &hAccel_s;
561
562#ifdef DUMMY
563 const unsigned long accelX_length = 0x00007568; // Load bearing definition???
564 /*
565 * Update sensor data with dummy values
566 * These arrays are defined in the files under /Data and are generated from
567 * past flight data binaries with srec_cat.
568 */
569 if (hDummyIdx < ACCELX_LENGTH - 1) {
570 // Shift in floating point values and add to processed accelerometer array
571 uint32_t tempX = (uint32_t)accelX[hDummyIdx + 1] << 16 | accelX[hDummyIdx];
572 uint32_t tempY = (uint32_t)accelY[hDummyIdx + 1] << 16 | accelY[hDummyIdx];
573 uint32_t tempZ = (uint32_t)accelZ[hDummyIdx + 1] << 16 | accelZ[hDummyIdx];
574 memcpy(&pAccel_s->accelData[0], &tempX, sizeof(float));
575 memcpy(&pAccel_s->accelData[1], &tempY, sizeof(float));
576 memcpy(&pAccel_s->accelData[2], &tempZ, sizeof(float));
577
578 // Back convert to raw data
579 uint16_t xRaw = (short)(pAccel_s->accelData[0] / pAccel_s->sensitivity);
580 uint16_t yRaw = (short)(pAccel_s->accelData[1] / pAccel_s->sensitivity);
581 uint16_t zRaw = (short)(pAccel_s->accelData[2] / pAccel_s->sensitivity);
582 pAccel_s->rawAccelData[0] = xRaw >> 8;
583 pAccel_s->rawAccelData[1] = xRaw;
584 pAccel_s->rawAccelData[2] = yRaw >> 8;
585 pAccel_s->rawAccelData[3] = yRaw;
586 pAccel_s->rawAccelData[4] = zRaw >> 8;
587 pAccel_s->rawAccelData[5] = zRaw;
588
589 // Shift in floating point values and add to processed gyroscope array
590 tempX = (uint32_t)gyroX[hDummyIdx + 1] << 16 | gyroX[hDummyIdx];
591 tempY = (uint32_t)gyroY[hDummyIdx + 1] << 16 | gyroY[hDummyIdx];
592 tempZ = (uint32_t)gyroZ[hDummyIdx + 1] << 16 | gyroZ[hDummyIdx];
593 memcpy(&gyro_s.gyroData[0], &tempX, sizeof(float));
594 memcpy(&gyro_s.gyroData[1], &tempY, sizeof(float));
595 memcpy(&gyro_s.gyroData[2], &tempZ, sizeof(float));
596
597 // Back convert to raw data
598 xRaw = (short)(gyro_s.gyroData[0] / gyro_s.sensitivity);
599 yRaw = (short)(gyro_s.gyroData[1] / gyro_s.sensitivity);
600 zRaw = (short)(gyro_s.gyroData[2] / gyro_s.sensitivity);
601 gyro_s.rawGyroData[0] = xRaw >> 8;
602 gyro_s.rawGyroData[1] = xRaw;
603 gyro_s.rawGyroData[2] = yRaw >> 8;
604 gyro_s.rawGyroData[3] = yRaw;
605 gyro_s.rawGyroData[4] = zRaw >> 8;
606 gyro_s.rawGyroData[5] = zRaw;
607
608 hDummyIdx += 2;
609 }
610#else
611 lAccel_s.update(&lAccel_s);
612 hAccel_s.update(&hAccel_s);
613 gyro_s.update(&gyro_s);
614#endif
615
616 // Add sensor data to dataframe
617 mem.append(&mem, HEADER_HIGHRES);
618 mem.appendBytes(&mem, pAccel_s->rawAccelData, KX134_1211_DATA_TOTAL);
619 mem.appendBytes(&mem, gyro_s.rawGyroData, A3G4250D_DATA_TOTAL);
620
621 // Only run calculations when enabled
622 EventBits_t uxBits = xEventGroupWaitBits(xTaskEnableGroup, GROUP_TASK_ENABLE_HIGHRES, pdFALSE, pdFALSE, blockTime);
623 if (uxBits & GROUP_TASK_ENABLE_HIGHRES) {
624 // Integrate attitude quaternion from rotations
625 Quaternion qDot;
626 Quaternion_init(&qDot);
627 qDot.fromEuler(
628 &qDot,
629 (float)(dt * gyro_s.gyroData[ROLL_INDEX]),
630 (float)(dt * gyro_s.gyroData[PITCH_INDEX]),
631 (float)(dt * gyro_s.gyroData[YAW_INDEX])
632 );
633 qRot = Quaternion_mul(&qRot, &qDot);
634 qRot.normalise(&qRot); // New attitude quaternion
635
636 // Apply rotation to z-axis unit vector
637 qRot.fRotateVector3D(&qRot, zUnit, vAttitude);
638
639 // Calculate tilt angle
640 // tilt = cos^-1(attitude · initial)
641 cosine = zUnit[0] * vAttitude[0] + zUnit[1] * vAttitude[1] + zUnit[2] * vAttitude[2];
642 tilt = acos(cosine) * 180 / M_PI;
643 }
644
645#ifdef DEBUG
648 if ((xSemaphoreTake(xUsbMutex, pdMS_TO_TICKS(0))) == pdTRUE) {
649 memset(HdebugStr, 100, sizeof(char));
650
651 snprintf(HdebugStr, 100, "[HDataAcq] %d\tAccel\tX: %.3f\tY: %.3f\tZ: %.3f\n\r", hDummyIdx / 2, pAccel_s->accelData[0], pAccel_s->accelData[1], pAccel_s->accelData[2]);
652 xMessageBufferSend(xUsbTxBuff, (void *)HdebugStr, 100, pdMS_TO_TICKS(0));
653
654 snprintf(HdebugStr, 100, "[HDataAcq] %d\tGyro\tX: %.3f\tY: %.3f\tZ: %.3f\n\r", hDummyIdx / 2, gyro_s.gyroData[0], gyro_s.gyroData[1], gyro_s.gyroData[2]);
655
656 xMessageBufferSend(xUsbTxBuff, (void *)HdebugStr, 100, pdMS_TO_TICKS(0));
657 xSemaphoreGive(xUsbMutex);
658 }
659#endif
660 }
661}

681void vDataAcquisitionL(void *argument) {
682 float dt = 0.020;
683 KalmanFilter kf;
684 KalmanFilter_init(&kf);
685
687 // Initialise filter parameters
688 float A[9] = {
689 1.0, dt, 0.5 * (dt * dt),
690 0.0, 1.0, dt,
691 0.0, 0.0, 1.0
692 };
693 kf.A.pData = A;
694 float Q[9] = {
695 99.52, 0.0, 0.0,
696 0.0, 1.42, 0.0,
697 0.0, 0.0, 6.27
698 };
699 kf.Q.pData = Q;
700 float R[4] = {
701 97.92, 0.0,
702 0.0, 0.61
703 };
704 kf.R.pData = R;
705 float P[9] = {
706 1, 0.0, 0.0,
707 0.0, 0.1, 0.0,
708 0.0, 0.0, 100.0
709 };
710 kf.P.pData = P;
711
712 // Initialise measurement matrix
713 arm_matrix_instance_f32 z;
714 float zData[2] = {0.0, 0.0};
715 arm_mat_init_f32(&z, 2, 1, zData);
716
717 TickType_t xLastWakeTime;
718 const TickType_t xFrequency = pdMS_TO_TICKS(20); // 50Hz
719 const TickType_t blockTime = pdMS_TO_TICKS(0);
720
721 for (;;) {
722 // Block until 20ms interval
723 vTaskDelayUntil(&xLastWakeTime, xFrequency);
724
725 // Update baro data
726#ifdef DUMMY
727 const unsigned long press_length = 0x00003A5C;
728 if (lDummyIdx < PRESS_LENGTH - 1) {
729 uint32_t tempPress = (uint32_t)press[lDummyIdx + 1] << 16 | press[lDummyIdx];
730 memcpy(&baro_s.press, &tempPress, sizeof(float));
731 lDummyIdx += 2;
732 }
733#else
734 baro_s.update(&baro_s);
735#endif
736
737 // Calculate altitude
738 altitude = 44330 * (1.0 - pow(baro_s.press / baro_s.groundPress, 0.1903));
739
740 // Add sensor data and barometer data to dataframe
741 mem.append(&mem, HEADER_LOWRES);
742 mem.appendBytes(&mem, baro_s.rawTemp, BMP581_DATA_SIZE);
743 mem.appendBytes(&mem, baro_s.rawPress, BMP581_DATA_SIZE);
744
745 // Only run calculations when enabled
746 EventBits_t uxBits = xEventGroupWaitBits(xTaskEnableGroup, GROUP_TASK_ENABLE_LOWRES, pdFALSE, pdFALSE, blockTime);
747 if (uxBits & GROUP_TASK_ENABLE_LOWRES) {
748 // Calculate state
749 z.pData[0] = altitude;
750 z.pData[1] = (cosine * 9.81 * pAccel_s->accelData[ZINDEX] - 9.81); // Acceleration measured in m/s^2
751 kf.update(&kf, &z);
752 velocity = kf.x.pData[1];
753
754 avgPress.append(&avgPress, baro_s.press);
755 avgVel.append(&avgVel, velocity);
756 }
757
758#ifdef DEBUG
761 if ((xSemaphoreTake(xUsbMutex, pdMS_TO_TICKS(0))) == pdTRUE) {
762 char debugStr[100];
763 snprintf(debugStr, 100, "[LDataAcq] %d\tBaro\tPressure: %.0f\n\r", lDummyIdx / 2, baro_s.press);
764 xMessageBufferSend(xUsbTxBuff, (void *)debugStr, 100, pdMS_TO_TICKS(10));
765 xSemaphoreGive(xUsbMutex);
766 }
767#endif
768 }
769}

800 );
