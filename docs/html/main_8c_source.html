169void vStateUpdate(void *argument) {
170 TickType_t xLastWakeTime;
171 const TickType_t xFrequency = pdMS_TO_TICKS(20); // 50Hz
172
173 unsigned int CANHigh = 0;
174 unsigned int CANLow = 0;
175 unsigned int id = 0;
176
177 uint8_t counter = 0;
178
179 for (;;) {
180 // Block until 20ms interval
181 vTaskDelayUntil(&xLastWakeTime, xFrequency);
182
183 // Emergency aerobrakes for excessive tilt
184 if (tilt >= 30.0f) {
185 // CAN payload for aerobrakes retract
186 CANHigh = 0x00000000;
187 CANLow = 0x00000000;
188 id = 0x602;
189 CAN_TX(1, 8, CANHigh, CANLow, id);
190 }
191
192 switch (currentState) {
193 case PRELAUNCH:
194 if (accel_s.accelData[ZINDEX] >= ACCEL_LAUNCH) {
195 GPIOB->ODR ^= 0X8000;
196 vTaskDelete(xUsbReceiveHandle);
197 xEventGroupSetBits(xTaskEnableGroup, GROUP_TASK_ENABLE_FLASH); // Enable flash
198 xEventGroupSetBits(xTaskEnableGroup, GROUP_TASK_ENABLE_HIGHRES); // Enable high resolution data acquisition
199 xEventGroupSetBits(xTaskEnableGroup, GROUP_TASK_ENABLE_LOWRES); // Enable low resolution data acquisition
200 currentState = LAUNCH;
201 }
202 break;
203 case LAUNCH:
204 avgVel.calculateMovingAverage(&avgVel, &avgVelCurrent);
205 // Send altitude to aerobrakes via CAN
206 CANHigh = 0x00000000;
207 CANLow = (unsigned int)altitude;
208 id = 0x601;
209 CAN_TX(1, 8, CANHigh, CANLow, id);
210 // Transition to motor burnout state on velocity decrease
211 if ((avgVelCurrent - avgVelPrevious) < 0) {
212 GPIOB->ODR ^= 0X8000;
213 currentState = COAST;
214 }
215 avgVelPrevious = avgVelCurrent;
216 break;
217 case COAST:
218 avgPress.calculateMovingAverage(&avgPress, &avgPressCurrent);
219 // Send altitude to aerobrakes via CAN
220 CANHigh = 0x00000000;
221 CANLow = (unsigned int) altitude;
222 id = 0x601;
223 CAN_TX(1, 8, CANHigh, CANLow, id);
224 // Transition to apogee state on three way vote of altitude, velocity, and tilt
225 // apogee is determined as two of three conditions evaluating true
226 if ((((avgPressCurrent - avgPressPrevious) > 0) + (tilt >= 90) + (velocity < 0.0f)) >= 2) {
227 GPIOB->ODR ^= 0X8000;
228 currentState = APOGEE;
229 // Send transmission to trigger apogee E-matches
230 }
231 avgPressPrevious = avgPressCurrent;
232 break;
233 case APOGEE:
234 GPIOB->ODR &= 0x8000;
235 if (altitude <= MAIN_ALTITUDE_METERS) {
236 currentState = DESCENT;
237 // Add descent event dataframe to buffer
238 }
239 break;
240 case DESCENT:
241 // Handle descent state actions
242 break;
243 }
244 }
245}

348void vUsbReceive(void *argument) {
349 const TickType_t timeout = pdMS_TO_TICKS(20);
350 uint8_t rxData;
351
352 for (;;) {
353 // Read byte from UART Rx buffer, skip loop if empty
354 if (!xStreamBufferReceive(xUsbRxBuff, (void *) &rxData, 1, timeout))
355 continue;
356
357 // Send byte back for display
358 usb.send(&usb, rxData);
359
360 // Process command and reset buffer on <Enter> input
361 if (rxData == CARRIAGE_RETURN) {
362 usb.print(&usb, "\n"); // Send newline back for display
363 usbRxBuff[usbRxBuffIdx-1] = '\0'; // Replace carriage return with null terminator
364 taskENTER_CRITICAL();
365 usbCommandParse(usbRxBuff); // Parse and execute command
366 taskEXIT_CRITICAL();
367 usbRxBuffIdx = 0; // Reset buffer
368 }
369 // Clear terminal on <Ctrl-c> input
370 else if (rxData == SIGINT) {
371 usbCommandParse((uint8_t *) "clear");
372 usbRxBuffIdx = 0;
373 }
374 // Erase character and move cursor backwards on <BS> input
375 else if (rxData == BACKSPACE) {
376 usb.print(&usb, " \b");
377 if (usbRxBuffIdx)
378 usbRxBuffIdx -= 2;
379 }
380 }

404void vDataAcquisitionH(void *argument) {
405 float dt = 0.002;
406
407 TickType_t xLastWakeTime;
408 const TickType_t xFrequency = pdMS_TO_TICKS(2); // 500Hz
409 const TickType_t blockTime = pdMS_TO_TICKS(0);
410
411
412 for (;;) {
413 // Block until 2ms interval
414 vTaskDelayUntil(&xLastWakeTime, xFrequency);
415
416 // Select which accelerometer to use
417 accel_s = (accel_s.accelData[ZINDEX] < 15) ? lAccel_s : hAccel_s;
418
419 // Update sensor data
420 #ifdef DUMMY
421 const unsigned long accelX_length = 0x00007568; // Load bearing definition???
422 if(hDummyIdx < ACCELX_LENGTH - 1) {
423 uint32_t tempX = (uint32_t)accelX[hDummyIdx+1] << 16 | accelX[hDummyIdx];
424 uint32_t tempY = (uint32_t)accelY[hDummyIdx+1] << 16 | accelY[hDummyIdx];
425 uint32_t tempZ = (uint32_t)accelZ[hDummyIdx+1] << 16 | accelZ[hDummyIdx];
426 memcpy(&accel_s.accelData[0], &tempX, sizeof(float));
427 memcpy(&accel_s.accelData[1], &tempY, sizeof(float));
428 memcpy(&accel_s.accelData[2], &tempZ, sizeof(float));
429 uint16_t xRaw = (short)(accel_s.accelData[0]/accel_s.sensitivity);
430 uint16_t yRaw = (short)(accel_s.accelData[1]/accel_s.sensitivity);
431 uint16_t zRaw = (short)(accel_s.accelData[2]/accel_s.sensitivity);
432 accel_s.rawAccelData[0] = xRaw >> 8;
433 accel_s.rawAccelData[1] = xRaw;
434 accel_s.rawAccelData[2] = yRaw >> 8;
435 accel_s.rawAccelData[3] = yRaw;
436 accel_s.rawAccelData[4] = zRaw >> 8;
437 accel_s.rawAccelData[5] = zRaw;
438
439 tempX = (uint32_t)gyroX[hDummyIdx+1] << 16 | gyroX[hDummyIdx];
440 tempY = (uint32_t)gyroY[hDummyIdx+1] << 16 | gyroY[hDummyIdx];
441 tempZ = (uint32_t)gyroZ[hDummyIdx+1] << 16 | gyroZ[hDummyIdx];
442 memcpy(&gyro_s.gyroData[0], &tempX, sizeof(float));
443 memcpy(&gyro_s.gyroData[1], &tempY, sizeof(float));
444 memcpy(&gyro_s.gyroData[2], &tempZ, sizeof(float));
445 xRaw = (short)(gyro_s.gyroData[0]/gyro_s.sensitivity);
446 yRaw = (short)(gyro_s.gyroData[1]/gyro_s.sensitivity);
447 zRaw = (short)(gyro_s.gyroData[2]/gyro_s.sensitivity);
448 gyro_s.rawGyroData[0] = xRaw >> 8;
449 gyro_s.rawGyroData[1] = xRaw;
450 gyro_s.rawGyroData[2] = yRaw >> 8;
451 gyro_s.rawGyroData[3] = yRaw;
452 gyro_s.rawGyroData[4] = zRaw >> 8;
453 gyro_s.rawGyroData[5] = zRaw;
454
455 hDummyIdx += 2;
456 }
457 #else
458 accel_s.update(&accel_s);
459 gyro_s.update(&gyro_s);
460 #endif
461
462 // Add sensor data to dataframe
463 mem.append(&mem, HEADER_HIGHRES);
464 mem.appendBytes(&mem, accel_s.rawAccelData, KX134_1211_DATA_TOTAL);
465 mem.appendBytes(&mem, gyro_s.rawGyroData, A3G4250D_DATA_TOTAL);
466
467 // Only run calculations when enabled
468 EventBits_t uxBits = xEventGroupWaitBits(xTaskEnableGroup, GROUP_TASK_ENABLE_HIGHRES, pdFALSE, pdFALSE, blockTime);
469 if (uxBits & GROUP_TASK_ENABLE_HIGHRES) {
470 // Integrate attitude quaternion from rotations
471 Quaternion qDot;
472 Quaternion_init(&qDot);
473 qDot.fromEuler(
474 &qDot,
475 (float)(dt * gyro_s.gyroData[ROLL_INDEX]),
476 (float)(dt * gyro_s.gyroData[PITCH_INDEX]),
477 (float)(dt * gyro_s.gyroData[YAW_INDEX])
478 );
479 qRot = Quaternion_mul(&qRot, &qDot);
480 qRot.normalise(&qRot); // New attitude quaternion
481
482 // Apply rotation to z-axis unit vector
483 qRot.fRotateVector3D(&qRot, zUnit, vAttitude);
484
485 // Calculate tilt angle
486 // tilt = cos^-1(attitude · initial)
487 cosine = zUnit[0] * vAttitude[0] + zUnit[1] * vAttitude[1] + zUnit[2] * vAttitude[2];
488 tilt = acos(cosine) * 180 / M_PI;
489 }
490 }
491}

497void vDataAcquisitionL(void *argument) {
498 float dt = 0.020;
499 KalmanFilter kf;
500 KalmanFilter_init(&kf);
501
502 // Initialise filter parameters
503 float A[9] = {
504 1.0, dt, 0.5 * (dt * dt),
505 0.0, 1.0, dt,
506 0.0, 0.0, 1.0
507 };
508 kf.A.pData = A;
509 float Q[9] = {
510 99.52, 0.0, 0.0,
511 0.0, 1.42, 0.0,
512 0.0, 0.0, 6.27
513 };
514 kf.Q.pData = Q;
515 float R[4] = {
516 97.92, 0.0,
517 0.0, 0.61
518 };
519 kf.R.pData = R;
520 float P[9] = {
521 1, 0.0, 0.0,
522 0.0, 0.1, 0.0,
523 0.0, 0.0, 100.0
524 };
525 kf.P.pData = P;
526
527 // Initialise measurement matrix
528 arm_matrix_instance_f32 z;
529 float zData[2] = {0.0, 0.0};
530 arm_mat_init_f32(&z, 2, 1, zData);
531
532 TickType_t xLastWakeTime;
533 const TickType_t xFrequency = pdMS_TO_TICKS(20); // 50Hz
534 const TickType_t blockTime = pdMS_TO_TICKS(0);
535
536 for (;;) {
537 // Block until 20ms interval
538 vTaskDelayUntil(&xLastWakeTime, xFrequency);
539
540 // Update baro data
541 #ifdef DUMMY
542 const unsigned long press_length = 0x00003A5C;
543 if(lDummyIdx < PRESS_LENGTH - 1) {
544 uint32_t tempPress = (uint32_t)press[lDummyIdx+1] << 16 | press[lDummyIdx];
545 memcpy(&baro_s.press, &tempPress, sizeof(float));
546 lDummyIdx += 2;
547 }
548 #else
549 baro_s.update(&baro_s);
550 #endif
551
552 // Calculate altitude
553 altitude = 44330 * (1.0 - pow(baro_s.press / baro_s.groundPress, 0.1903));
554
555 // Add sensor data and barometer data to dataframe
556 mem.append(&mem, HEADER_LOWRES);
557 mem.appendBytes(&mem, baro_s.rawTemp, BMP581_DATA_SIZE);
558 mem.appendBytes(&mem, baro_s.rawPress, BMP581_DATA_SIZE);
559
560 // Only run calculations when enabled
561 EventBits_t uxBits = xEventGroupWaitBits(xTaskEnableGroup, GROUP_TASK_ENABLE_LOWRES, pdFALSE, pdFALSE, blockTime);
562 if (uxBits & GROUP_TASK_ENABLE_LOWRES) {
563 // Calculate state
564 z.pData[0] = altitude;
565 z.pData[1] = (cosine * 9.81 * accel_s.accelData[ZINDEX] - 9.81); // Acceleration measured in m/s^2
566 kf.update(&kf, &z);
567 velocity = kf.x.pData[1];
568
569 avgPress.append(&avgPress, baro_s.press);
570 avgVel.append(&avgVel, velocity);
571 }
572 }
573}
