26void BMP581_init(BMP581 *baro, GPIO_TypeDef *port, unsigned long cs, float tempSensitivity, float pressSensitivity) {
27 SPI_init(&baro->base, SENSOR_BARO, SPI1, MODE8, port, cs);
28 baro->tempSensitivity = tempSensitivity;
29 baro->pressSensitivity = pressSensitivity;
30 baro->update = BMP581_update;
31 baro->readTemp = BMP581_readTemp;
32 baro->readRawTemp = BMP581_readRawTemp;
33 baro->processRawTemp = BMP581_processRawTemp;
34 baro->readPress = BMP581_readPress;
35 baro->readRawPress = BMP581_readRawPress;
36 baro->processRawPress = BMP581_processRawPress;
37
38 const uint32_t superDelay = 0xFFFF;
39 volatile uint8_t counter = 0;
40
41 // Wait for the spefified period - need to wait for 2ms here.
42 for (uint32_t i = 0; i < superDelay; i++) {
43 counter++;
44 }
45
46 BMP581_writeRegister(baro, BMP581_ODR_CFG, BMP581_ODR_CFG_PWR | BMP581_ODR_CFG_DEEP_DIS);
47 uint8_t OSRCFG = BMP581_readRegister(baro, BMP581_OSR_CFG);
48 BMP581_writeRegister(baro, BMP581_OSR_CFG, (BMP581_OSR_CFG_RESERVED & OSRCFG) | BMP581_OSR_CFG_PRESS_EN);
49
50 // Set ground pressure reading on init
51 baro->readPress(baro, &baro->groundPress);
52}

154void BMP581_writeRegister(BMP581 *baro, uint8_t address, uint8_t data) {
155 SPI spi = baro->base;
156
157 spi.port->ODR &= ~spi.cs;
158
159 // Send read command and address
160 uint8_t payload = address & 0x7F; // Load payload with address and read command
161 spi.transmit(&spi, payload); // Transmit payload
162 spi.transmit(&spi, data); // Transmit dummy data and read response data
163
164 spi.port->ODR |= spi.cs;
165}
166
167uint8_t BMP581_readRegister(BMP581 *baro, uint8_t address) {
168 uint8_t response = 0;
169 SPI spi = baro->base;
170
171 spi.port->ODR &= ~spi.cs;
172
173 // Send read command and address
174 uint8_t payload = address | 0x80; // Load payload with address and read command
175 response = spi.transmit(&spi, payload); // Transmit payload
176 response = spi.transmit(&spi, 0xFF); // Transmit dummy data and read response data
177
178 spi.port->ODR |= spi.cs;
179
180 return response;
181}
